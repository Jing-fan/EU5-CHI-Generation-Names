# ==============================================================================
# 宗室字辈系统 - 核心脚本效果引擎 (zibei_effects.txt)
# ==============================================================================
# 模块总览：
#   零、皇室字辈方案初始化 (zibei_init_dynasty_scheme_effect)
#       - 明式 (1/2)：始祖 = 99，父 = 98，已出生子辈 = 100（新出生的男丁取木字旁单字名），孙辈起 = 1..20
#       - 清式与通谱 (3..7)：始祖 = 20，父 = 19，已出生子辈起 = 1..20
#   一、出生代际传递与方案分发 (zibei_inherit_and_name_effect)
#       - 代际递推：parent.gen + 1；21 -> 1（满 20 代回环），101 -> 1（明式孙辈入第 1 字辈）
#   二、七套字辈方案主函数（含明式始祖之子辈单字木字旁专属函数）
#   三、智能分流 (zibei_smart_pick_effect)：区分性别、皇室近支/旁系及百年偏旁水位线保护
#   四、独立无参滑动冷却抽号池 (13 个 100 字男池 + 1 个 200 字女池，防止近期重名)
#   五、静态键值赋名映射（对应 zibei_names_l_simp_chinese.yml 及 zibei_wood_001..100）
# ==============================================================================

# ==========================================================
# 全局参数配置（修改此处数值即可调整冷却长度与重置周期）
# 冷却步数是指在多少步内该字库的字不能再次被抽取，这个值越小重名的可能越高; 而太大则会导致命名随机性下降。
# 水位线计数器是指在多少年后重置偏旁字库的使用次数，考虑到一个字库的重复使用中间应当间隔4代人，100年是一个很保守的数值
# ==========================================================
@zibei_close_branch_cd = 95        # 近支皇子占用偏旁字的冷却步数（100字库锁95步）
@zibei_distant_branch_cd = 45      # 旁系宗室占用偏旁字的冷却步数（45步，优先释放）
@zibei_general_cd = 90             # 通用男字库（100字）的冷却步数
@zibei_female_cd = 180             # 女性专属字库（200字）的冷却步数
@zibei_watermark_reset_years = 100 # 偏旁字库水位线计数器的重置周期（100年）

# ==========================================================
# 零、 皇室字辈方案初始化（决议与事件调用入口）
# ==========================================================
zibei_init_dynasty_scheme_effect = {
	# 记录本次正在确立的字辈方案编号，供紧接弹出的 zibei_events.3 独立配置该方案的女性赐名礼制
	set_variable = { name = zibei_pending_scheme value = $SCHEME$ }
	ruler ?= {
		save_scope_as = zibei_founder
		set_variable = { name = zibei_was_emperor value = 1 }
		set_variable = { name = zibei_scheme value = $SCHEME$ }

		if = {
			limit = {
				OR = {
					var:zibei_scheme = 1
					var:zibei_scheme = 2
				}
			}
			# 明式（五行相生）：始祖代际设为 99，其父为 98，其子辈为 100（新出生的儿子取木字旁单字名），
			# 自孙辈起（100 + 1 = 101 -> 1）正式进入二十字辈与火/土/金/水/木五行轮转
			set_variable = { name = zibei_generation value = 99 }
			father ?= {
				set_variable = { name = zibei_father_of_emperor value = 1 }
				set_variable = { name = zibei_scheme value = $SCHEME$ }
				set_variable = { name = zibei_generation value = 98 }
			}
			# 为已出生的同宗族直系子嗣（100=木字旁单字辈）、孙辈（第1代）、曾孙辈（第2代）初始化代数
			every_child = {
				limit = { dynasty ?= scope:zibei_founder.dynasty }
				set_variable = { name = zibei_scheme value = $SCHEME$ }
				set_variable = { name = zibei_generation value = 100 }
				every_child = {
					limit = { prev = { is_female = no } }
					set_variable = { name = zibei_scheme value = $SCHEME$ }
					set_variable = { name = zibei_generation value = 1 }
					every_child = {
						limit = { prev = { is_female = no } }
						set_variable = { name = zibei_scheme value = $SCHEME$ }
						set_variable = { name = zibei_generation value = 2 }
					}
				}
			}
		}
		else = {
			# 清式与通谱：始祖代际设为 20（正整数非零，确保引擎持久化存储），
			# 其子出生时 +1 变为 21 自动回环为第 1 代，直接自子辈起按字辈命名
			set_variable = { name = zibei_generation value = 20 }
			father ?= {
				set_variable = { name = zibei_father_of_emperor value = 1 }
				set_variable = { name = zibei_scheme value = $SCHEME$ }
				set_variable = { name = zibei_generation value = 19 }
			}
			# 为已出生的同宗族直系子嗣（第1代）、孙辈（第2代）、曾孙辈（第3代）初始化代数
			every_child = {
				limit = { dynasty ?= scope:zibei_founder.dynasty }
				set_variable = { name = zibei_scheme value = $SCHEME$ }
				set_variable = { name = zibei_generation value = 1 }
				every_child = {
					limit = { prev = { is_female = no } }
					set_variable = { name = zibei_scheme value = $SCHEME$ }
					set_variable = { name = zibei_generation value = 2 }
					every_child = {
						limit = { prev = { is_female = no } }
						set_variable = { name = zibei_scheme value = $SCHEME$ }
						set_variable = { name = zibei_generation value = 3 }
					}
				}
			}
		}
	}
	# 为国内其他同宗族成员（如兄弟、叔伯、堂兄弟等）兜底初始化字辈方案，防止兄终弟及或旁支继位时断代
	if = {
		limit = { exists = scope:zibei_founder.dynasty }
		every_character = {
			limit = {
				dynasty ?= scope:zibei_founder.dynasty
				OR = {
					NOT = { has_variable = zibei_scheme }
					var:zibei_scheme = 99
				}
			}
			set_variable = { name = zibei_scheme value = $SCHEME$ }
			if = {
				limit = {
					father ?= {
						has_variable = zibei_generation
					}
				}
				set_variable = { name = zibei_generation value = father.var:zibei_generation }
				change_variable = { name = zibei_generation add = 1 }
				if = {
					limit = {
						OR = {
							var:zibei_generation = 21
							var:zibei_generation = 101
						}
					}
					set_variable = { name = zibei_generation value = 1 }
				}
			}
			else = {
				if = {
					limit = {
						OR = {
							var:zibei_scheme = 1
							var:zibei_scheme = 2
						}
					}
					set_variable = { name = zibei_generation value = 99 }
				}
				else = {
					set_variable = { name = zibei_generation value = 20 }
				}
			}
		}
	}
}

# ==========================================================
# 一、 出生代际传递与方案分发入口
# ==========================================================
zibei_inherit_and_name_effect = {
	if = {
		limit = {
			father ?= {
				has_variable = zibei_scheme
				has_variable = zibei_generation
				var:zibei_scheme >= 1
				var:zibei_scheme <= 7
			}
		}
		father = {
			save_scope_as = zibei_parent
		}
	}
	else_if = {
		limit = {
			mother ?= {
				has_variable = zibei_scheme
				has_variable = zibei_generation
				var:zibei_scheme >= 1
				var:zibei_scheme <= 7
			}
		}
		mother = {
			save_scope_as = zibei_parent
		}
	}

	if = {
		limit = { exists = scope:zibei_parent }
		set_variable = {
			name = zibei_scheme
			value = scope:zibei_parent.var:zibei_scheme
		}
		set_variable = {
			name = zibei_generation
			value = scope:zibei_parent.var:zibei_generation
		}
		change_variable = {
			name = zibei_generation
			add = 1
		}
		# 20代满环（21 -> 1）或明式始祖之子辈下一代（101 -> 1），统一进入/回到第 1 字辈
		if = {
			limit = {
				OR = {
					var:zibei_generation = 21
					var:zibei_generation = 101
				}
			}
			set_variable = { name = zibei_generation value = 1 }
		}

		# 分支1：明式始祖同辈（gen = 99，如始祖之父在世再生子），仅静默继承代际变量，不改名
		if = {
			limit = { var:zibei_generation = 99 }
		}
		# 分支2：明式始祖之子辈（gen = 100）：男丁先从木字旁字库取单字名，女儿保留常规随机名字
		else_if = {
			limit = { var:zibei_generation = 100 }
			if = {
				limit = {
					is_female = no
					OR = {
						var:zibei_scheme = 1
						var:zibei_scheme = 2
					}
				}
				zibei_apply_ming_son_wood_name_effect = yes
			}
		}
		# 分支3：正式进入二十字辈（gen = 1..20）
		# 若为男性，或该字辈方案开启了“男女皆按字辈赐名”，则执行字辈赐名；
		# 若未开启女眷赐名，女性子嗣仅静默继承代际变量，保持常规随机名字
		else_if = {
			limit = {
				OR = {
					is_female = no
					zibei_scheme_includes_females_trigger = yes
				}
			}
			if = {      limit = { var:zibei_scheme = 1 } zibei_apply_ming_donggong_name_effect = yes }
			else_if = { limit = { var:zibei_scheme = 2 } zibei_apply_ming_yanwang_name_effect = yes }
			else_if = { limit = { var:zibei_scheme = 3 } zibei_apply_qing_name_effect = yes }
			else_if = { limit = { var:zibei_scheme = 4 } zibei_apply_other_mao_name_effect = yes }
			else_if = { limit = { var:zibei_scheme = 5 } zibei_apply_other_kong_name_effect = yes }
			else_if = { limit = { var:zibei_scheme = 6 } zibei_apply_other_qian_name_effect = yes }
			else_if = { limit = { var:zibei_scheme = 7 } zibei_apply_other_yan_name_effect = yes }
		}
	}
}

# 二、 七套字辈方案主函数
# 由于EU5引擎不支持动态传参，故每套字辈方案都需要独立的语句，以下的重复代码为无奈之举
# 不过性能上并没有太大的开销，轮询完全足够应对，考虑到可能的拓展，没有在这里做更复杂的优化
# ==========================================================

# 0. 明式·始祖之子辈专属：从木字旁字库（100字）抽取单字名（以应“木生火”之始）
zibei_apply_ming_son_wood_name_effect = {
	if = {
		limit = { zibei_is_imperial_close_branch_trigger = yes }
		set_local_variable = { name = zibei_cd_len value = @zibei_close_branch_cd }
	}
	else = {
		set_local_variable = { name = zibei_cd_len value = @zibei_distant_branch_cd }
	}
	zibei_roll_pool_wood = yes
	zibei_assign_wood_single_name = yes
}

zibei_smart_pick_effect = {
	# 若为女性子嗣，直接从200字女性雅字库中抽取第3个字（与字辈字拼接），不消耗男性偏旁名额
	if = {
		limit = { is_female = yes }
		set_local_variable = { name = zibei_cd_len value = @zibei_female_cd }
		zibei_roll_pool_female = yes
		zibei_assign_female_name = { ZIBEI = $ZIBEI$ }
	}
	else = {
		# 1. 每100年定时重置偏旁字库的水位线计数器
		if = {
			limit = {
				OR = {
					NOT = { has_global_variable = zibei_reset_year_$POOL$ }
					current_year >= global_var:zibei_reset_year_$POOL$
				}
			}
			set_global_variable = { name = zibei_reset_year_$POOL$ value = current_year }
			change_global_variable = { name = zibei_reset_year_$POOL$ add = @zibei_watermark_reset_years }
			set_global_variable = { name = zibei_gen_used_$POOL$ value = 1 }
		}

		# 2. 决定使用偏旁字库还是通用字库，并设定对应的 local_var:zibei_cd_len
		if = {
			limit = { zibei_is_imperial_close_branch_trigger = yes }
			set_local_variable = { name = zibei_use_general value = 1 }
			set_local_variable = { name = zibei_cd_len value = @zibei_close_branch_cd }
		}
		else = {
			random_list = {
				80 = {
					trigger = { global_var:zibei_gen_used_$POOL$ < 80 }
					modifier = {
						add = {
							value = global_var:zibei_gen_used_$POOL$
							multiply = -1
						}
					}
					set_local_variable = { name = zibei_use_general value = 1 }
					set_local_variable = { name = zibei_cd_len value = @zibei_distant_branch_cd }
				}
				10 = {
					modifier = { add = global_var:zibei_gen_used_$POOL$ }
					set_local_variable = { name = zibei_use_general value = 2 }
					set_local_variable = { name = zibei_cd_len value = @zibei_general_cd }
				}
			}
		}

		# 3. 执行无参抽号函数 + 轻量赋名映射
		if = {
			limit = { local_var:zibei_use_general = 1 }
			change_global_variable = { name = zibei_gen_used_$POOL$ add = 1 }
			zibei_roll_pool_$POOL$ = yes
			zibei_assign_name = { POOL = $POOL$ ZIBEI = $ZIBEI$ }
		}
		else = {
			zibei_roll_pool_general = yes
			zibei_assign_name = { POOL = general ZIBEI = $ZIBEI$ }
		}
	}
}
